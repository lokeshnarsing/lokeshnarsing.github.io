@Service
public class LoadManager {

  // Share site power equally across ports
  public Map<Port, Double> distribute(Site site) {
    List<Port> active = site.activePorts();
    double share = site.powerLimitKw()
                 / active.size();

    return active.stream().collect(toMap(
        p -> p,
        p -> Math.min(share, p.maxKw())));
  }
}
